We define the accumulated state as
\begin{equation}
  \label{eq:accumulation}
  x_j = \sum_{k=i}^{j-1} A_k x_k.
\end{equation}
The stability condition is
\begin{equation}
  \label{eq:stability}
  \mathbb{E}\|x_k\|^2 \geq C\rho^k\|x_0\|^2.
\end{equation}
